cmake_minimum_required(VERSION 3.20)
project(aion_m1 CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(aion_m1 INTERFACE)
target_include_directories(aion_m1 INTERFACE ${CMAKE_SOURCE_DIR}/include)
target_compile_features(aion_m1 INTERFACE cxx_std_20)
option(AION_M1_BUILD_TESTS "Build tests" ON)
if(AION_M1_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
